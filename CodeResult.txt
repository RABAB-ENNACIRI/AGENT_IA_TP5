import math

a=6
b=11
c=-6
D = b*b - 4*a*c
sqrtD = math.sqrt(D)
root1 = (-b + sqrtD) / (2*a)
root2 = (-b - sqrtD) / (2*a)

print("Exact roots: (-11 ± sqrt(265)) / 12")
print("Numeric approximations:")
print(root1)
print(root2)
